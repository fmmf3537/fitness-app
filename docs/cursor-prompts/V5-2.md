MAX_DAILY_INPUT_CHARS = 8000  # 单次提炼输入截断（防对话爆量）
MAX_MEMORY_PER_DISTILL = 3   # 单次提炼产出 ≤3 条摘要

def _query_today_messages(session, day: date) -> tuple[list, list]:
    """查当天新增 report_chat / coach_chat 消息（按 created_at >= 当天 00:00）。
    返回 (report_chat_msgs, coach_chat_msgs) 两个 list[(role, content), ...]，
    每组为单条报告/独立聊天的对话轮次。
    """

def _format_messages_for_distill(messages: list) -> str:
    """把 [(role, content), ...] 格式化成"用户：…/教练：…"的多行文本，输入上限 8k 字截断。"""

def _call_distill_llm(text: str, *, chat_fn) -> list[dict]:
    """调 LLM 把对话提炼成 [{"summary": "...", "tags": "..."}, ...]，≤3 条。
    chat_fn 用于测试注入；默认走 adapters.llm.chat(purpose='memory_distill')。
    输出 schema 校验：必须是 list[dict]，每条必须有 summary 与 tags。
    失败抛 LLMError 由调用方 try/except 兜底。
    """

def distill_messages_to_memory(session, messages, source, ref_id, *, chat_fn=None) -> int:
    """把单条对话（report_chat 或 coach_chat）提炼入 coach_memory。
    幂等：按 (source, ref_id) 去重——已存在则跳过（通过 CoachMemory 的 source + ref_report_id/ref_chat_id 联合查询）。
    返回新插入条数（0 表示去重跳过或 LLM 失败）。
    """

def distill_daily_memory(session, day: date, *, chat_fn=None) -> dict:
    """每日批提炼：跑 _query_today_messages + 对每条对话调 distill_messages_to_memory。
    任何异常（LLM 失败 / DB 失败）均写 JobRun(job_name='memory_distill') 但不抛。
    返回 {"date", "conversations_processed", "memories_added", "errors": [...]}。
    """

def build_query_tags(workout_dict: dict, report_type: str) -> list[str]:
    """聚合本次上下文的 tag：movement names + body_part + garmin activity type + report type。
    去重 + 去空。
    """

def compose_memory_section_for(
    session,
    workout_dict: dict,
    report_type: str,
) -> str:
    """包装：调 build_query_tags + search_memory + list_preferences(active_only=True) + build_memory_section。
    供 5 入口调用：返回值即 memory_section 参数。
    空数据时返回 ""（build_memory_section 已有 AC5 行为）。
    L3 本切片传 None（V5-3 才接长期统计查询）。
    """
